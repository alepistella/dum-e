cmake_minimum_required(VERSION 3.22)
project(sensor_reset_component_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(gz-sim8 REQUIRED COMPONENTS rendering)
find_package(gz-rendering8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-sensors8 REQUIRED COMPONENTS boundingbox_camera camera depth_camera gpu_lidar lidar rgbd_camera segmentation_camera thermal_camera wide_angle_camera)
add_library(review_rendering SHARED
 src/rendering/MarkerManager.cc src/rendering/RenderUtil.cc
 src/rendering/SceneManager.cc src/rendering/WrenchVisualizer.cc)
target_link_libraries(review_rendering PUBLIC gz-sim8::gz-sim8 gz-rendering8::gz-rendering8 PRIVATE gz-plugin2::register)
set_target_properties(review_rendering PROPERTIES OUTPUT_NAME gz-sim8-rendering SOVERSION 8 VERSION 8.15.0 LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/overlay/lib")
add_library(review_sensors SHARED src/systems/sensors/Sensors.cc)
target_link_libraries(review_sensors PUBLIC review_rendering gz-sim8::gz-sim8 gz-common5::gz-common5 gz-rendering8::gz-rendering8 gz-plugin2::register)
foreach(c boundingbox_camera camera depth_camera gpu_lidar lidar rgbd_camera segmentation_camera thermal_camera wide_angle_camera)
 target_link_libraries(review_sensors PRIVATE gz-sensors8::${c})
endforeach()
set_target_properties(review_sensors PROPERTIES OUTPUT_NAME gz-sim8-sensors-system SOVERSION 8 VERSION 8.15.0 LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/overlay/plugins")
add_custom_command(TARGET review_sensors POST_BUILD COMMAND ${CMAKE_COMMAND} -E create_symlink $<TARGET_FILE_NAME:review_sensors> "${CMAKE_BINARY_DIR}/overlay/plugins/libgz-sim-sensors-system.so")
